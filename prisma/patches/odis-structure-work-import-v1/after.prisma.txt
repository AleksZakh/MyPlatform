model DiagnosticStructureWork {
  id Int @id @default(autoincrement())
  contractStructureId Int
  structureId Int
  workId Int
  workCodeSnapshot String @db.VarChar(100)
  workNameSnapshot String @db.VarChar(300)
  criteriaSnapshot String @db.Text
  catalogHoursSnapshot Decimal @db.Decimal(12, 4)
  hours Decimal? @db.Decimal(14, 4)
  note String @default("") @db.Text
  planStart DateTime? @db.Date
  planFinish DateTime? @db.Date
  planHours Decimal? @db.Decimal(14, 4)
  planExecutor String @default("") @db.Text
  factStart DateTime? @db.Date
  factFinish DateTime? @db.Date
  factHours Decimal? @db.Decimal(14, 4)
  factExecutor String @default("") @db.Text
  status String @default("") @db.VarChar(100)
  warning String @default("") @db.Text
  legacyCanonicalId Int? @unique
  legacyBusinessHash String? @unique @db.VarChar(64)
  legacyUserWorkId Int?
  legacySource Json?
  legacyPersonalSource Json?
  importSourceHash String? @db.VarChar(64)
  importUserSourceHash String? @db.VarChar(64)
  importedById Int?
  importedAt DateTime?
  legacyOrigins DiagnosticStructureWorkOrigin[]
  version Int @default(1)
  createdById Int?
  updatedById Int?
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt
  deletedAt DateTime?
  deletedById Int?
  contractStructure DiagnosticContractStructure @relation(fields:[contractStructureId,structureId],references:[id,structureId],onDelete:Restrict,map:"ds_work_contract_link_fk")
  work DiagnosticWork @relation(fields:[workId],references:[id],onDelete:Restrict)
  @@index([contractStructureId, deletedAt])
  @@index([workId])
  @@map("diagnostic_structure_works")
}

model DiagnosticStructureWorkOrigin {
  legacyId Int @id
  assignmentId Int
  sourceIdRaw String @db.Text
  sourceCode String @db.VarChar(100)
  sourceHash String @db.VarChar(64)
  sourceFileHash String @db.VarChar(64)
  assignment DiagnosticStructureWork @relation(fields:[assignmentId],references:[id],onDelete:Restrict)
  @@index([assignmentId])
  @@map("diagnostic_structure_work_origins")
}
