User:   spaceGroups SpaceGroupUser[]
DomainGroup:   spaceGroups SpaceGroupDomain[]

Добавить модели один раз:

model SpaceGroup {
  id Int @id @default(autoincrement())
  name String @unique @db.VarChar(255)
  isActive Boolean @default(true)
  users SpaceGroupUser[]
  domainGroups SpaceGroupDomain[]
  permissions SpaceGroupPermission[]
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt
  @@map("space_groups")
}
model SpaceGroupUser {
  groupId Int
  userId Int
  group SpaceGroup @relation(fields: [groupId], references: [id], onDelete: Cascade)
  user User @relation(fields: [userId], references: [id], onDelete: Cascade)
  @@id([groupId, userId])
  @@index([userId])
  @@map("space_group_users")
}
model SpaceGroupDomain {
  groupId Int
  domainGroupId Int
  group SpaceGroup @relation(fields: [groupId], references: [id], onDelete: Cascade)
  domainGroup DomainGroup @relation(fields: [domainGroupId], references: [id], onDelete: Cascade)
  @@id([groupId, domainGroupId])
  @@index([domainGroupId])
  @@map("space_group_domains")
}
model SpaceGroupPermission {
  id Int @id @default(autoincrement())
  groupId Int
  resourceId Int
  action AccessAction
  grantedByLogin String? @db.VarChar(150)
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt
  group SpaceGroup @relation(fields: [groupId], references: [id], onDelete: Cascade)
  resource AccessResource @relation(fields: [resourceId], references: [id], onDelete: Cascade)
  @@unique([groupId, resourceId, action])
  @@index([resourceId, action])
  @@map("space_group_permissions")
}
